// Number of cells in each direction
Nx := 512
Ny := 512
Nz := 1

// Units
um := 1e-6
mm := 1e-3

// Parameters of computational domain (meters)
size_x := 5000 * um
size_y := 5000 * um
size_z := 10 * um

// Parameters of a single cell in each direction (meters)
cell_size_x := size_x / Nx
cell_size_y := size_y / Ny
cell_size_z := size_z / Nz

SetGridsize(Nx, Ny, Nz)
SetCellsize(cell_size_x, cell_size_y, cell_size_z)

SetGeom(ImageShape("./geometry/9_rings_4_waveguids(without_PML).png"))


pad := 250 * um // Distance between antenna and edge of PML

waveguide_width := 300 * um // Width of the waveguide
waveguide_length := 5000 * um // Length of the waveguide
waveguide_pad := 810 * um // Distance between waveguide and edge of domain

// Parameters of the micro strip (antenna)
micro_strip_width := 20 * um
micro_strip_length := waveguide_width // Length of the micro strip is equal to the width of the waveguide


// YIG parameters
alpha = 0.00001
Aex = 3.614e-12
Msat = 1.39e5 // (1853 * 1000) / (4 * pi)


// ABL setup: Exponential gradient 
alpha_min := 0.00001
alpha_max := 1.0

abl_levels := 32 // Number of regions for ABL
abl_size := 300 * um // Length of ABL region

abl_layer_size := abl_size / abl_levels // Size for a single layer of ABL region

// Area of the domain where ABL is not applied
base_rectangle_x := size_x - 2 * abl_size
base_rectangle_y := size_y - 2 * abl_size
base_rectangle_z := size_z

for level := 1; level <= abl_levels; level++ {
    abl_layer := Cuboid(
                    base_rectangle_x + level * abl_layer_size * 2,
                    base_rectangle_y + level * abl_layer_size * 2,
                    base_rectangle_z).sub(Cuboid(
                        base_rectangle_x + (level - 1) * abl_layer_size * 2,
                        base_rectangle_y + (level - 1) * abl_layer_size * 2,
                        base_rectangle_z))

    // Exponential growth of alpha from inner to outer layer
    layer_coefficient := alpha_min * pow(alpha_max / alpha_min, 1.0 * level / abl_levels)

    DefRegion(level, abl_layer)
    alpha.SetRegion(level, layer_coefficient)
}

Save(alpha)


B_stat := 1200e-4
bb := 90 // Angle of the wave propagation in degrees
m = uniform(cos(bb * pi / 180), sin(bb * pi / 180), 0)
B_ext = vector(B_stat * cos(bb * pi / 180), B_stat * sin(bb * pi / 180), 0) // If bb = 90, then wage propagates along the y-axis, and the external field is along the x-axis

OutputFormat = OVF2_TEXT

Relax()
SaveAs(m, "ground_state_m")
SaveAs(B_demag, "ground_state_B_ext")


// Amp_sinc := 0.001
Amp := 8.0e-4

t0 := 50.0e-9
f0 := $frequency
Fs := 2 * f0

SetSolver(5) // This call is redundant because the Dormand-Prince method is the default solver in MuMax3
MaxDt = 1 / Fs
MinDt = 1e-15
MaxErr = 1.0e-8

/*
Antenna coordinates
Note: these coordinates define the center of the antenna in the coordinate plain, 
that is matched with coordinates (indices) of cells in the simulation domain 
(down left corner - (0, 0, 0), upper right corner - (Nx-1, Ny-1, Nz-1)).
*/

// Fist antenna - the down left corner of the domain
$first_antenna_open
antenna_x := abl_size + pad + micro_strip_width / 2
antenna_y := waveguide_pad + micro_strip_length / 2
$first_antenna_close

// Second antenna - the up right corner of the domain
$second_antenna_open
antenna1_x := size_x - (abl_size + pad + micro_strip_width / 2)
antenna1_y := size_y - (waveguide_pad + micro_strip_length / 2)
$second_antenna_close

// Third antenna - the up left corner of the domain
$third_antenna_open
antenna2_x := abl_size + pad + micro_strip_width / 2
antenna2_y := size_y - (waveguide_pad + micro_strip_length / 2)
$third_antenna_close

// Fourth antenna - the down right corner of the domain
$fourth_antenna_open
antenna3_x := size_x - (abl_size + pad + micro_strip_width / 2)
antenna3_y := waveguide_pad + micro_strip_length / 2
$fourth_antenna_close

mask := NewSlice(3, Nx, Ny, 1)

$first_antenna_open
for i := floor((antenna_x - micro_strip_width / 2) / cell_size_x);
    i <= floor((antenna_x + micro_strip_width / 2) / cell_size_x);
    i++ {
        for j := floor((antenna_y - micro_strip_length / 2) / cell_size_y);
            j <= floor((antenna_y + micro_strip_length / 2) / cell_size_y);
            j++ {
                //B_x := pow(cos(pi * antenna_x), 2)
                //B_y := pow(cos(pi * antenna_y), 2)
                //B_y := 1.0 - pow(2.0 * abs(antenna_y - 0.5), 3)

                current_cell := Index2Coord(i, j, 0)
                x := current_cell.X() + size_x / 2
                y := current_cell.Y() + size_y / 2

                B_x := pow(cos(pi * abs(x - antenna_x) / micro_strip_width), 2)
                B_y := pow(cos(pi * abs(y - antenna_y) / micro_strip_length), 2)

                B := B_x * B_y
                mask.Set(2, i, j, 0, B) // MSSW
        }
}
$first_antenna_close

$second_antenna_open
for i := floor((antenna1_x - micro_strip_width / 2) / cell_size_x);
    i <= floor((antenna1_x + micro_strip_width / 2) / cell_size_x);
    i++ {

                x := current_cell.X() + size_x / 2
                y := current_cell.Y() + size_y / 2

                B_x := pow(cos(pi * abs(x - antenna1_x) / micro_strip_width), 2)
                B_y := pow(cos(pi * abs(y - antenna1_y) / micro_strip_length), 2)

                B := B_x * B_y
                mask.Set(2, i, j, 0, B)
        }
}
$second_antenna_close

$third_antenna_open
for i := floor((antenna2_x - micro_strip_width / 2) / cell_size_x);
    i <= floor((antenna2_x + micro_strip_width / 2) / cell_size_x);
    i++ {
        for j := floor((antenna2_y - micro_strip_length / 2) / cell_size_y);
            j <= floor((antenna2_y + micro_strip_length / 2) / cell_size_y);
            j++ {
                current_cell := Index2Coord(i, j, 0)
                x := current_cell.X() + size_x / 2
                y := current_cell.Y() + size_y / 2

                B_x := pow(cos(pi * abs(x - antenna2_x) / micro_strip_width), 2)
                B_y := pow(cos(pi * abs(y - antenna2_y) / micro_strip_length), 2)

                B := B_x * B_y
                mask.Set(2, i, j, 0, B)
        }
}
$third_antenna_close

$fourth_antenna_open
for i := floor((antenna3_x - micro_strip_width / 2) / cell_size_x);
    i <= floor((antenna3_x + micro_strip_width / 2) / cell_size_x);
    i++ {
        for j := floor((antenna3_y - micro_strip_length / 2) / cell_size_y);

                mask.Set(2, i, j, 0, B)
        }
}
$fourth_antenna_close


B_ext.Add(mask, Amp * sin(2 * pi * f0 * t))
